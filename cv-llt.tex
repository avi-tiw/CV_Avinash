\DeclareSortingTemplate{custom}{
  % 1. Primary sort: Put entries WITH a pubstate (e.g. unpublished, submitted) last,
  %    and entries WITHOUT a pubstate (published papers) first.
  \sort{
    \field{pubstate}
    \literal{0}
  }
  % 2. Secondary sort: Year in descending order (newest to oldest)
  \sort[direction=descending]{
    \field{sortyear}
    \field{year}
    \literal{9999}
  }
  % 3. Tertiary sort: Month in descending order
  \sort[direction=descending]{
    \field{month}
    \literal{99}
  }
}
