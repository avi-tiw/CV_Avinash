\begin{rubric}{Research Background}
\entry*[] \textbf{Data Analysis:} Parameter estimation, Fisher matrix analysis, and hierarchical inference. \textbf{Waveform Modeling:} Analytical modeling of line-of-sight acceleration and velocity waveforms. \textbf{Probing Merger Environments:} developing methods for probing CBC environments on a single-event basis. \textbf{Characterizing GW signals:} Developing model-independent tests to characterize the GW signals. \textbf{Cosmology:} Early universe cosmology driven by the synergies between GW and 21-cm global signal observations.
    
\end{rubric}
